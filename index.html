create table public.saved_searches ( id bigint generated by default as identity primary key, user_id uuid not null references auth.users(id) on delete cascade, search_name text not null, location text not null, created_at timestamptz default now() ); alter table public.saved_searches enable row level security; grant usage on schema public to authenticated; grant select, insert, update, delete on table public.saved_searches to authenticated; create policy "Users can view their own saved searches" on public.saved_searches for select to authenticated using (auth.uid() = user_id); create policy "Users can create their own saved searches" on public.saved_searches for insert to authenticated with check (auth.uid() = user_id);
